   114→export function ProfileDialog({
   115→  username,
   116→  onClose,
   117→  onDm,
   118→  onFriend,
   119→  canModerate,
   120→  onMod,
   121→  friendState,
   122→}: {
   123→  username: string | null;
   124→  onClose: () => void;
   125→  onDm: (u: string) => void;
   126→  onFriend: (u: string) => void;
   127→  canModerate: boolean;
   128→  onMod: (u: string) => void;
   129→  friendState?: "none" | "friends" | "pending" | "self";
   130→}) {
   131→  const [info, setInfo] = useState<{
   132→    user: ChatUserSummary & { bio?: string; xp?: number; coins?: number; streak?: number; role?: string };
   133→    messages: number;
   134→  } | null>(null);
   135→
   136→  useEffect(() => {
   137→    if (!username) return;
   138→    setInfo(null);
   139→    void emitAck<{
   140→      user: ChatUserSummary & { bio?: string; xp?: number; coins?: number; streak?: number; role?: string };
   141→      messages: number;
   142→    }>("profile:get", username).then((r) => {
   143→      const d = r as typeof info;
   144→      if (d && d.user) setInfo(d);
   145→    });
   146→  }, [username]);
   147→
   148→  const u = info?.user;
   149→
   150→  return (
   151→    <Dialog open={!!username} onOpenChange={(o) => !o && onClose()}>
   152→      <DialogContent className="bg-[#050505] border-[#1f1f1f] text-white max-w-sm">
   153→        <DialogHeader>
   154→          <DialogTitle className="flex items-center gap-2 text-base">
   155→            <UserRound className="size-4" /> {username}
   156→          </DialogTitle>
   157→          {u?.role && u.role !== "member" && (
   158→            <DialogDescription className="text-[11px] font-mono uppercase">
   159→              {u.role}
   160→            </DialogDescription>
   161→          )}
   162→        </DialogHeader>
   163→        {!u ? (
   164→          <p className="text-[13px] text-[#666] py-4 boot-pulse w-fit">loading…</p>
   165→        ) : (
   166→          <div className="space-y-3">
   167→            <div className="grid grid-cols-4 gap-2 text-center">
   168→              {[
   169→                ["XP", u.xp ?? 0],
   170→                ["Coins", u.coins ?? 0],
   171→                ["Msgs", info?.messages ?? 0],
   172→                ["Streak", u.streak ?? 0],

   179→                    {label}
   180→                  </div>
   181→                </div>
   182→              ))}
   183→            </div>

   185→              <p className="text-[12px] text-[#8f8f8f]">
   186→                <span className="text-[#666]">status:</span> {u.status}
   187→              </p>
   188→            )}
   189→            {u.bio && <p className="text-[13px] text-[#c9c9c9]">{u.bio}</p>}

   191→              {friendState !== "self" && (

   193→                  size="sm"

   207→                  <UserPlus className="size-3.5 mr-1" /> Add friend

   212→                  size="sm"

   458→/* ---------------- soundboard ---------------- */
   459→
   460→const SOUNDS: { id: string; label: string; freqs: number[]; type: OscillatorType; dur: number }[] = [
   461→  { id: "ding", label: "ding", freqs: [880], type: "sine", dur: 0.4 },
   462→  { id: "buzz", label: "buzz", freqs: [110], type: "sawtooth", dur: 0.5 },
   463→  { id: "blip", label: "blip", freqs: [660, 990], type: "square", dur: 0.18 },
   464→  { id: "tada", label: "tada", freqs: [523, 659, 784, 1047], type: "triangle", dur: 0.7 },
   465→  { id: "boom", label: "boom", freqs: [60], type: "sine", dur: 0.8 },
   466→  { id: "ping", label: "ping", freqs: [1318], type: "sine", dur: 0.3 },
   467→];
   468→
   469→export function playLocalSound(id: string) {
   470→  const def = SOUNDS.find((s) => s.id === id);
   471→  if (!def) return;
   472→  try {
   473→    const AC =
   474→      window.AudioContext ||
   475→      (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
   476→    const ctx = new AC();
   477→    let t = ctx.currentTime;
   478→    for (const f of def.freqs) {
   479→      const osc = ctx.createOscillator();
   480→      const g = ctx.createGain();
   481→      osc.type = def.type;
   482→      osc.frequency.value = f;
   483→      g.gain.setValueAtTime(0.18, t);
   484→      g.gain.exponentialRampToValueAtTime(0.001, t + def.dur);
   485→      osc.connect(g).connect(ctx.destination);
   486→      osc.start(t);
   487→      osc.stop(t + def.dur + 0.02);
   488→      t += def.dur / def.freqs.length;
   489→    }
   490→    setTimeout(() => void ctx.close(), (def.dur + 0.3) * 1000);
   491→  } catch {
   492→    /* audio unavailable */
   493→  }
   494→}

   523→            >

   533→/* ---------------- friends dialog ---------------- */
   534→
   535→interface FriendRow {
   536→  id: string;
   537→  status: string;
   538→  user: { id: string; username: string; displayName: string | null; status: string } | null;
   539→}
   540→
   541→export function FriendsDialog({
   542→  open,
   543→  onOpenChange,
   544→  onDm,
   545→  refreshKey,
   546→}: {
   547→  open: boolean;
   548→  onOpenChange: (o: boolean) => void;
   549→  onDm: (u: string) => void;
   550→  refreshKey: number;
   551→}) {
   552→  const [rows, setRows] = useState<FriendRow[]>([]);
   553→  const [addName, setAddName] = useState("");
   554→  const [msg, setMsg] = useState("");
   555→
   556→  const load = useCallback(() => {
   557→    void emitAck<FriendRow[]>("friends:list").then((r) => {
   558→      const d = r as FriendRow[];
   559→      if (Array.isArray(d)) setRows(d);
   560→    });
   561→  }, []);
   562→
   563→  useEffect(() => {
   564→    if (open) load();
   565→  }, [open, load, refreshKey]);
   566→
   567→  const friends = rows.filter((r) => r.status === "accepted");
   568→  const incoming = rows.filter((r) => r.status === "pending" && r.user?.id);
   569→  const sent = rows.filter((r) => r.status === "pending" && !r.user);
   570→
   571→  const request = async () => {
   572→    const name = addName.trim();
   573→    if (!name) return;
   574→    const r = await emitAck<{ ok?: boolean; error?: string }>("friends:request", name);
   575→    if (r.error) setMsg(r.error);
   576→    else {
   577→      setMsg("Request sent.");
   578→      setAddName("");
   579→      load();
   580→    }
   581→    setTimeout(() => setMsg(""), 2500);
   582→  };
   583→
   584→  const respond = async (id: string, accept: boolean) => {
   585→    await emitAck("friends:respond", { id, accept });
   586→    load();
   587→  };
   588→
   589→  const remove = async (id: string) => {
   590→    await emitAck("friends:remove", id);
   591→    load();
   592→  };
   593→
   594→  const Row = ({ r, pending }: { r: FriendRow; pending?: boolean }) => (
   595→    <div className="flex items-center gap-2 py-1.5">
   596→      <span className="size-1.5 rounded-full bg-white shrink-0" aria-hidden />
   597→      <span className="text-[13px] text-white truncate flex-1">
   598→        {r.user ? r.user.displayName || r.user.username : "unknown"}
   599→      </span>
   600→      {pending ? (
   601→        <>
   602→          <Button size="sm" variant="outline" onClick={() => void respond(r.id, true)} className="h-7 px-2 border-[#2a2a2a] text-white hover:bg-[#101010]">
   603→            Accept
   604→          </Button>
   605→          <Button size="sm" variant="ghost" onClick={() => void respond(r.id, false)} className="h-7 px-2 text-[#888] hover:text-white">
   606→            Decline
   607→          </Button>
   608→        </>
   609→      ) : (
   610→        <>
   611→          <Button size="sm" variant="ghost" onClick={() => r.user && onDm(r.user.username)} className="h-7 px-2 text-[#999] hover:text-white">
   612→            <MessageSquare className="size-3.5" />
   613→          </Button>
   614→          <Button size="sm" variant="ghost" aria-label="Remove friend" onClick={() => void remove(r.id)} className="h-7 px-2 text-[#888] hover:text-white">
   615→            <UserMinus className="size-3.5" />
   616→          </Button>
   617→        </>
   618→      )}
   619→    </div>
   620→  );
   621→
   622→  return (
   623→    <Dialog open={open} onOpenChange={onOpenChange}>
   624→      <DialogContent className="bg-[#050505] border-[#1f1f1f] text-white max-w-sm">
   625→        <DialogHeader>
   626→          <DialogTitle className="flex items-center gap-2 text-base">
   627→            <UserPlus className="size-4" /> Friends
   628→          </DialogTitle>
   629→          <DialogDescription className="text-xs text-[#7a7a7a]">
   630→            Add friends to DM them more easily.
   631→          </DialogDescription>
   632→        </DialogHeader>
   633→        <div className="flex gap-2">
   634→          <Input
   635→            value={addName}
   636→            onChange={(e) => setAddName(e.target.value)}
   637→            onKeyDown={(e) => e.key === "Enter" && void request()}
   638→            placeholder="Username"
   639→            aria-label="Friend username"
   640→            className="bg-[#050505] border-[#1f1f1f] text-white placeholder:text-[#555]"
   641→          />
   642→          <Button onClick={() => void request()} size="sm" className="bg-white text-black hover:bg-[#e0e0e0] shrink-0">
   643→            Add
   644→          </Button>
   645→        </div>
   646→        {msg && <p className="text-[12px] text-[#999]">{msg}</p>}
   647→        {incoming.length > 0 && (
   648→          <section>
   649→            <p className="mono-label mb-1">Requests ({incoming.length})</p>
   650→            {incoming.map((r) => (
   651→              <Row key={r.id} r={r} pending />
   652→            ))}
   653→          </section>
   654→        )}
   655→        {sent.length > 0 && (
   656→          <section>
   657→            <p className="mono-label mb-1">Sent</p>
   658→            {sent.map((r) => (
   659→              <div key={r.id} className="flex items-center gap-2 py-1.5">
   660→                <span className="size-1.5 rounded-full border border-[#555] shrink-0" aria-hidden />
   661→                <span className="text-[13px] text-[#999] flex-1">pending…</span>
   662→              </div>
   663→            ))}
   664→          </section>
   665→        )}
   666→        <section>
   667→          <p className="mono-label mb-1">Friends ({friends.length})</p>
   668→          {friends.length === 0 && (
   669→            <p className="text-[13px] text-[#666]">No friends added yet.</p>
   670→          )}
   671→          {friends.map((r) => (
   672→            <Row key={r.id} r={r} />
   673→          ))}
   674→        </section>
   675→      </DialogContent>
   676→    </Dialog>
   677→  );
   678→}

   682→export function CreateChannelDialog({
   683→  open,
   684→  onOpenChange,
   685→  onCreated,
   686→}: {
   687→  open: boolean;
   688→  onOpenChange: (o: boolean) => void;
   689→  onCreated: (id: string, name: string) => void;

   760→          </div>

   824→        >
   825→          Join
   826→        </Button>
