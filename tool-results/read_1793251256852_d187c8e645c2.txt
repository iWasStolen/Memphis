   247→function MessageBubble({
   248→  m,
   249→  me,
   250→  onProfile,
   251→  onDelete,
   252→  onVote,
   253→}: {
   254→  m: ChatMessage;
   255→  me: SessionUser | null;
   256→  onProfile: (u: string) => void;
   257→  onDelete: (id: string) => void;
   258→  onVote: (messageId: string, optionIndex: number) => void;
   259→}) {
   260→  const poll = m.poll;
   261→  const ended = poll && poll.endsAt < Date.now();
   262→  const totalVotes = poll
   263→    ? poll.options.reduce((a, o) => a + o.voters.length, 0)
   264→    : 0;
   265→  const isStaff = me && ["owner", "admin", "mod"].includes(me.role);
   266→
   267→  if (m.isSystem || m.deleted) {
   268→    return (
   269→      <div className="text-center py-1">
   270→        <span className="text-[11.5px] text-[#666] bg-[#0a0a0a] border border-[#161616] rounded-full px-3 py-1 inline-block">
   271→          {m.deleted ? "message deleted" : m.content}
   272→        </span>
   273→      </div>
   274→    );
   275→  }
   276→
   277→  return (
   278→    <div className="group flex gap-2.5 px-1 py-1 hover:bg-[#0a0a0a] rounded-md">
   279→      <button
   280→        onClick={() => onProfile(m.username)}
   281→        aria-label={`View ${m.username}`}
   282→        className="size-8 shrink-0 rounded-full border border-[#262626] flex items-center justify-center text-[11px] font-mono text-white hover:border-white transition-colors"
   283→      >
   284→        {(m.displayName || m.username).slice(0, 2).toUpperCase()}
   285→      </button>
   286→      <div className="min-w-0 flex-1">
   287→        <div className="flex items-baseline gap-2">
   288→          <button
   289→            onClick={() => onProfile(m.username)}
   290→            className="text-[13px] text-white font-medium hover:underline"
   291→          >
   292→            {m.displayName || m.username}
   293→          </button>
   294→          {m.role && m.role !== "member" && (
   295→            <span className="text-[10px] font-mono uppercase border border-[#2a2a2a] rounded px-1 text-[#aaa]">
   296→              {m.role}
   297→            </span>
   298→          )}
   299→          <span className="text-[10.5px] text-[#555]">
   300→            {new Date(m.createdAt).toLocaleTimeString([], {
   301→              hour: "2-digit",
   302→              minute: "2-digit",
   303→            })}
   304→          </span>
   305→        </div>
   306→
   307→        <p className="text-[13.5px] text-[#d5d5d5] whitespace-pre-wrap break-words mt-0.5">
   308→          {m.content}
   309→        </p>
   310→
   311→        {poll && (
   312→          <div className="mt-2 border border-[#1a1a1a] rounded-lg p-2.5 max-w-sm">
   313→            <p className="text-[12px] text-[#999] flex items-center gap-1.5 mb-2">
   314→              <Vote className="size-3" /> {poll.question}
   315→              <span className="ml-auto text-[#555]">
   316→                {ended ? "final" : `${Math.max(0, Math.round((poll.endsAt - Date.now()) / 60000))}m left`}
   317→              </span>
   318→            </p>
   319→            {poll.options.map((o, i) => {
   320→              const pct = totalVotes ? Math.round((o.voters.length / totalVotes) * 100) : 0;
   321→              const voted = o.voters.includes(me?.id ?? "");
   322→              return (
   323→                <button
   324→                  key={i}
   325→                  disabled={!!ended}
   326→                  onClick={() => onVote(m.id, i)}
   327→                  className={`w-full text-left mb-1 last:mb-0 rounded border px-2 py-1.5 text-[12.5px] transition-colors ${
   328→                    voted
   329→                      ? "border-white text-white"
   330→                      : "border-[#222] text-[#bbb] hover:border-[#444]"
   331→                  }`}
   332→                >
   333→                  <span className="flex justify-between">
   334→                    <span>{o.text}</span>
   335→                    <span className="text-[#777] font-mono">
   336→                      {pct}% ({o.voters.length})
   337→                    </span>
   338→                  </span>
   339→                  <span className="block h-1 bg-[#1a1a1a] rounded-full mt-1 overflow-hidden">
   340→                    <span
   341→                      className="block h-full bg-white transition-all"
   342→                      style={{ width: `${pct}%` }}
   343→                    />
   344→                  </span>
   345→                </button>
   346→              );
   347→            })}
   348→          </div>
   349→        )}
   350→      </div>
   351→
   352→      {(isStaff || m.userId === me?.id) && (
   353→        <button
   354→          onClick={() => onDelete(m.id)}
   355→          aria-label="Delete message"
   356→          className="opacity-0 group-hover:opacity-100 transition-opacity p-1.5 text-[#666] hover:text-white self-start"
   357→        >
   358→          <Trash2 className="size-3.5" />
   359→        </button>
   360→      )}
   361→    </div>
   362→  );
   363→}

   410→  useEffect(() => {
   411→    const stored = getToken();

   413→    if (!stored) {

   479→    };
   480→    const onInvited = () => {

   482→        (r) => {

   485→        }

   495→    s.on("chat:updated", onUpdated);
   496→    s.on("chat:deleted", onDeleted);
   497→    s.on("chat:cleared", onCleared);
   498→    s.on("chat:notice", onNotice);
   499→    s.on("chat:typing", onTyping);
   500→    s.on("presence:list", onPresence);
   501→    s.on("user:update", onUserUpdate);
   502→    s.on("friends:incoming", onIncoming);
   503→    s.on("dm:refresh", onDmRefresh);
   504→    s.on("sound:play", onSound);
   505→    s.on("group:invited", onInvited);
   506→
   507→    if (s.connected) onConnect();
   508→
   509→    return () => {
   510→      s.off("auth:ok", onAuthOk);
   511→      s.off("auth:error", onAuthError);
   512→      s.off("connect", onConnect);
   513→      s.off("disconnect", onDisconnect);
   514→      s.off("chat:message", onMessage);
   515→      s.off("chat:updated", onUpdated);
   516→      s.off("chat:deleted", onDeleted);
   517→      s.off("chat:cleared", onCleared);
   518→      s.off("chat:notice", onNotice);
   519→      s.off("chat:typing", onTyping);
   520→      s.off("presence:list", onPresence);
   521→      s.off("user:update", onUserUpdate);
   522→      s.off("friends:incoming", onIncoming);
   523→      s.off("dm:refresh", onDmRefresh);
   524→      s.off("sound:play", onSound);
   525→      s.off("group:invited", onInvited);
   526→    };
   527→  }, [user, room, setUser]);

   529→  function roomKeyOf(r: Room | null): string | null {
   530→    if (!r) return null;
