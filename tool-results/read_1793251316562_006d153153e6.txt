    86→function filterMessage(text: string): { text: string; severe: boolean } {
    87→  const cfg = settings();
    88→  const words = cfg.wordFilter
    89→    .split(",")
    90→    .map((w) => w.trim().toLowerCase())
    91→    .filter(Boolean);
    92→  let out = text;
    93→  for (const w of words) {
    94→    const re = new RegExp(
    95→      w.replace(/[.*+?^${}()|[\]\\]/g, "\\$&"),
    96→      "gi"
    97→    );
    98→    out = out.replace(re, "•".repeat(Math.max(3, w.length)));
    99→  }
   100→  const norm = normalize(text);
   101→  const severe = SEVERE.some((w) => norm.includes(w));
   102→  return { text: out, severe };
   103→}

   167→function fetchMessages(where: string, arg: unknown, limit = 60, before?: number) {
   168→  const conds = [where, before ? "createdAt < ?" : ""].filter(Boolean).join(" AND ");
   169→  const args = [arg, ...(before ? [before] : [])];
   170→  const rows = db
   171→    .query(
   172→      `SELECT m.*, u.username, u.displayName, u.avatarUrl, u.role FROM Message m
   173→       JOIN User u ON m.userId = u.id
   174→       WHERE ${conds} AND m.deleted = 0
   175→       ORDER BY m.createdAt DESC LIMIT ?`
   176→    )
   177→    .all(...args, limit);
   178→  return rows.reverse().map(msgRow);
   179→}

   210→function floodCheck(u: UserRow): string | null {
   211→  const cfg = settings();
   212→  if (!cfg.automodEnabled) return null;
   213→  const now = Date.now();
   214→  const arr = (msgTimes.get(u.id) ?? []).filter((t) => now - t < cfg.floodSecs * 1000);
   215→  arr.push(now);
   216→  msgTimes.set(u.id, arr);
   217→  if (arr.length > cfg.floodMsgs)
   218→    return `flood (${arr.length} messages in ${cfg.floodSecs}s)`;
   219→  return null;
   220→}
   221→
   222→function repeatCheck(u: UserRow, content: string): string | null {
   223→  const cfg = settings();
   224→  if (!cfg.automodEnabled || cfg.sameMsgLimit <= 0) return null;
   225→  const arr = (lastMessages.get(u.id) ?? []).slice(-(cfg.sameMsgLimit - 1));
   226→  arr.push(content);
   227→  lastMessages.set(u.id, arr);
   228→  if (arr.filter((m) => m === content).length >= cfg.sameMsgLimit)
   229→    return `repeated messages`;
   230→  return null;
   231→}

   879→      case "unlockChannel": {

   978→  if (u.coins < bet) {
   979→    return cb?.({ error: "Not enough coins for that bet" });
   980→  }
   981→
   982→  if (cmd === "flip") {
   983→    const side = (parts[2] ?? "heads").toLowerCase();
   984→    if (!["heads", "tails"].includes(side)) return cb?.({ error: "Usage: /flip <bet> heads|tails" });
   985→    addCoins(u.id, -bet);
   986→    const result = Math.random() < 0.5 ? "heads" : "tails";
   987→    if (side === result) {
   988→      addCoins(u.id, bet * 2);
   989→      sys(`🪙 ${result} — ${u.username} won ${bet} coins`);
   990→    } else {
   991→      sys(`🪙 ${result} — ${u.username} lost ${bet} coins`);
   992→    }
   993→    emitUserUpdate(io, u.id);
   994→    return cb?.({ ok: true });
   995→  }
   996→  if (cmd === "slots") {
   997→    addCoins(u.id, -bet);
   998→    const reels = slotsSpin();
   999→    const jackpot = reels[0] === reels[1] && reels[1] === reels[2];
  1000→    const two = reels[0] === reels[1] || reels[1] === reels[2] || reels[0] === reels[2];
  1001→    let win = 0;
  1002→    if (jackpot) win = bet * 10;
  1003→    else if (two) win = Math.floor(bet * 1.5);
  1004→    if (win) addCoins(u.id, win);
  1005→    sys(
  1006→      `${reels.join(" ")} — ${u.username} ${win ? `won ${win} coins` : `lost ${bet} coins`}`
  1007→    );
  1008→    emitUserUpdate(io, u.id);
  1009→    return cb?.({ ok: true });
  1010→  }
  1011→  if (cmd === "balance" || cmd === "bal") {
  1012→    return cb?.({ ok: true, notice: `You have ${u.coins} coins.` });
  1013→  }
  1014→  return cb?.({ error: "Commands: /flip <bet> heads|tails · /slots <bet> · /bal" });
  1015→}

  1017→function handleCounting(socket: Socket, io: Server, u: UserRow, ch: { id: string; kind: string }, content: string, cb: (r: unknown) => void) {
  1018→  const num = parseInt(content.trim(), 10);
  1019→  const lastRow = db
  1020→    .query(
  1021→      `SELECT m.content, m.userId FROM Message m WHERE m.channelId = ? AND m.isSystem = 0 AND m.deleted = 0
  1022→       ORDER BY m.createdAt DESC LIMIT 1`
  1023→    )
  1024→    .get(ch.id) as { content: string; userId: string } | undefined;
  1025→  const expected = lastRow ? (parseInt(lastRow.content, 10) || 0) + 1 : 1;
  1026→  if (!Number.isFinite(num) || String(num) !== content.trim()) {
  1027→    return cb?.({ error: "Numbers only in this channel" });
  1028→  }
  1029→  if (lastRow && lastRow.userId === u.id) {
  1030→    return cb?.({ error: "You can't count twice in a row" });
  1031→  }
  1032→  if (num !== expected) {
  1033→    const id = insertMessage(u.id, content, { channelId: ch.id });
  1034→    io.to("ch:" + ch.id).emit("chat:message", messageWithJoins(id));
  1035→    const sysId = insertMessage(
  1036→      u.id,
  1037→      `❌ ${u.username} broke the chain at ${expected - 1}. Counting restarts at 1.`,
  1038→      { channelId: ch.id, isSystem: true }
  1039→    );
  1040→    io.to("ch:" + ch.id).emit("chat:message", messageWithJoins(sysId));
  1041→    return cb?.({ ok: true });
  1042→  }
  1043→  const id = insertMessage(u.id, content, { channelId: ch.id });
  1044→  io.to("ch:" + ch.id).emit("chat:message", messageWithJoins(id));
  1045→  if (num % 100 === 0) {
  1046→    addCoins(u.id, 10);
  1047→    emitUserUpdate(io, u.id);
  1048→    const sysId = insertMessage(u.id, `🏆 ${num}! ${u.username} earned 10 coins.`, {

  1052→    io.to("ch:" + ch.id).emit("chat:message", messageWithJoins(sysId));
