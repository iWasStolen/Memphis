   131→/* split a message into TTS-sized chunks (API limit is 1024 chars) */
   132→function splitForTts(text: string, max = 950): string[] {
   133→  const clean = text
   134→    .replace(/```[\s\S]*?```/g, " — code block omitted — ")
   135→    .replace(/!\[[^\]]*\]\([^)]*\)/g, " — image — ");
   136→  const sentences = clean.match(/[^.!?\n]+[.!?]*\s*/g) ?? [clean];
   137→  const chunks: string[] = [];
   138→  let cur = "";
   139→  for (const s of sentences) {
   140→    if ((cur + s).length > max) {
   141→      if (cur.trim()) chunks.push(cur);
   142→      if (s.length > max) {
   143→        for (let i = 0; i < s.length; i += max) chunks.push(s.slice(i, i + max));
   144→        cur = "";
   145→      } else cur = s;
   146→    } else cur += s;
   147→  }
   148→  if (cur.trim()) chunks.push(cur);
   149→  return chunks;
   150→}

   152→/* 16 kHz mono PCM WAV encoder (for voice input) */
   153→function encodeWav(buffer: AudioBuffer, targetRate = 16000): ArrayBuffer {
   154→  const src = buffer.getChannelData(0);
   155→  const ratio = buffer.sampleRate / targetRate;
   156→  const length = Math.max(1, Math.floor(src.length / ratio));
   157→  const data = new Float32Array(length);
   158→  for (let i = 0; i < length; i++) {
   159→    const pos = i * ratio;
   160→    const i0 = Math.floor(pos);
   161→    const i1 = Math.min(i0 + 1, src.length - 1);
   162→    const frac = pos - i0;
   163→    data[i] = src[i0] * (1 - frac) + src[i1] * frac;
   164→  }
   165→  const out = new ArrayBuffer(44 + length * 2);
   166→  const view = new DataView(out);
   167→  const wstr = (off: number, s: string) => {
   168→    for (let i = 0; i < s.length; i++) view.setUint8(off + i, s.charCodeAt(i));
   169→  };
   170→  wstr(0, "RIFF");
   171→  view.setUint32(4, 36 + length * 2, true);
   172→  wstr(8, "WAVE");
   173→  wstr(12, "fmt ");
   174→  view.setUint32(16, 16, true);
   175→  view.setUint16(20, 1, true);
   176→  view.setUint16(22, 1, true);
   177→  view.setUint32(24, targetRate, true);
   178→  view.setUint32(28, targetRate * 2, true);
   179→  view.setUint16(32, 2, true);
   180→  view.setUint16(34, 16, true);
   181→  wstr(36, "data");
   182→  view.setUint32(40, length * 2, true);
   183→  let off = 44;
   184→  for (let i = 0; i < length; i++, off += 2) {
   185→    const s = Math.max(-1, Math.min(1, data[i]));
   186→    view.setInt16(off, s < 0 ? s * 0x8000 : s * 0x7fff, true);
   187→  }
   188→  return out;
   189→}

   193→function Md({ text }: { text: string }) {
   194→  return (
   195→    <div className="memphis-md text-[13.5px] leading-relaxed break-words">
   196→      <ReactMarkdown
   197→        remarkPlugins={[remarkGfm]}
   198→        components={{
   199→          code(props) {
   200→            const { className, children } = props;
   201→            const match = /language-(\w+)/.exec(className || "");
   202→            const code = String(children ?? "").replace(/\n$/, "");
   203→            if (!match && !code.includes("\n")) {
   204→              return (
   205→                <code className="bg-[#161616] border border-[#262626] rounded px-1 py-0.5 text-[12px] font-mono">
   206→                  {code}
   207→                </code>
   208→              );
   209→            }
   210→            return (
   211→              <code className={className}>
   212→                <SyntaxHighlighter
   213→                  style={oneDark as Record<string, Record<string, string>>}
   214→                  language={match ? match[1] : "text"}
   215→                  PreTag="div"
   216→                  customStyle={{
   217→                    borderRadius: "10px",
   218→                    border: "1px solid #1f1f1f",
   219→                    fontSize: "12px",
   220→                    margin: "8px 0",
   221→                    background: "#0a0a0a",
   222→                  }}
   223→                >
   224→                  {code}
   225→                </SyntaxHighlighter>
   226→              </code>
   227→            );
   228→          },
   229→          a(props) {
   230→            const { href, children } = props;
   231→            return (
   232→              <a href={href} target="_blank" rel="noreferrer noopener" className="u-link text-white underline underline-offset-2">
   233→                {children}
   234→              </a>
   235→            );
   236→          },
   237→          img(props) {
   238→            const { src, alt } = props;
   239→            if (!src || typeof src !== "string") return null;
   240→            return (
   241→              <img
   242→                src={src}
   243→                alt={alt ?? ""}
   244→                loading="lazy"
   245→                onClick={() => window.open(src, "_blank")}
   246→                className="max-w-full max-h-[420px] rounded-lg border border-[#222] my-2 cursor-zoom-in hover:border-[#444] transition-colors"
   247→              />
   248→            );
   249→          },
   250→        }}
   251→      >
   252→        {text}
   253→      </ReactMarkdown>
   254→    </div>
   255→  );
   256→}

   536→    if (speaking?.id === m.id) {

   544→      const chunks = splitForTts(m.content);
   545→      let first = true;
   546→      for (const chunk of chunks) {
   547→        if (speechTokenRef.current !== token) return;
   548→        const r = await fetch("/api/ai/tts", {
   549→          method: "POST",
   550→          headers: { "content-type": "application/json" },

   552→        });

   554→        if (first) {
   555→          setSpeaking({ id: m.id, loading: false });
   556→          first = false;
