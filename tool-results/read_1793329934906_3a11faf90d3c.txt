    83→    83→/* Pump a z-ai completion (raw SSE bytes, strings, or parsed chunk
    84→    84→   objects), collecting and forwarding reasoning_content + content.
    85→    85→   Returns everything the call produced. */
    86→    86→async function pumpStream(
    87→    87→  completion: unknown,
    88→    88→  onReasoning: (t: string) => void,
    89→    89→  onDelta: (t: string) => void
    90→    90→): Promise<{ content: string; reasoning: string }> {
    91→    91→  const decoder = new TextDecoder();
    92→    92→  let rawBuf = "";
    93→    93→  let content = "";
    94→    94→  let reasoning = "";
    95→    95→
    96→    96→  const handleObj = (obj: {
    97→    97→    choices?: {
    98→    98→      delta?: { content?: string; reasoning_content?: string };
    99→    99→      message?: { content?: string; reasoning_content?: string };
   100→   100→    }[];
   101→   101→  }) => {
   102→   102→    const thinkDelta =
   103→   103→      obj?.choices?.[0]?.delta?.reasoning_content ??
   104→   104→      obj?.choices?.[0]?.message?.reasoning_content ??
   105→   105→      "";
   106→   106→    if (thinkDelta) {
   107→   107→      reasoning += thinkDelta;
   108→   108→      onReasoning(thinkDelta);
   109→   109→    }
   110→   110→    const delta =
   111→   111→      obj?.choices?.[0]?.delta?.content ??
   112→   112→      obj?.choices?.[0]?.message?.content ??
   113→   113→      "";
   114→   114→    if (delta) {
   115→   115→      content += delta;
   116→   116→      onDelta(delta);
   117→   117→    }
   118→   118→  };
   119→   119→
   120→   120→  const consumeLine = (line: string) => {
   121→   121→    const t = line.trim();
   122→   122→    if (!t.startsWith("data:")) return;
   123→   123→    const payload = t.slice(5).trim();
   124→   124→    if (!payload || payload === "[DONE]") return;
   125→   125→    try {
   126→   126→      handleObj(JSON.parse(payload));
   127→   127→    } catch {
   128→   128→      /* partial json — ignore */
   129→   129→    }
   130→   130→  };
   131→   131→
   132→   132→  const consumeBuf = () => {
   133→   133→    const lines = rawBuf.split("\n");
   134→   134→    rawBuf = lines.pop() ?? "";
   135→   135→    for (const line of lines) consumeLine(line);
   136→   136→  };
   137→   137→
   138→   138→  const iterable = completion as AsyncIterable<unknown> | null;
   139→   139→  if (iterable && typeof iterable[Symbol.asyncIterator] === "function") {
   140→   140→    for await (const chunk of iterable) {
   141→   141→      if (chunk instanceof Uint8Array) {
   142→   142→        rawBuf += decoder.decode(chunk, { stream: true });
   143→   143→        consumeBuf();
   144→   144→      } else if (typeof chunk === "string") {
   145→   145→        rawBuf += chunk;
   146→   146→        consumeBuf();
   147→   147→      } else if (chunk && typeof chunk === "object") {
   148→   148→        handleObj(chunk as Parameters<typeof handleObj>[0]);
   149→   149→      }
   150→   150→    }
   151→   151→    rawBuf += decoder.decode();
   152→   152→    consumeBuf();
   153→   153→  } else {
   154→   154→    handleObj(
   155→   155→      (completion as { choices?: { message?: { content?: string; reasoning_content?: string } }[] }) ?? {}
   156→   156→    );
   157→   157→  }
   158→   158→  return { content, reasoning };
   159→   159→}

   446→   446→            } else {

   450→   450→            }
   451→   451→          } catch (e) {

   453→   453→              message: `Image generation failed (${e instanceof Error ? e.message : "unknown error"}). Try rephrasing.`,
   454→   454→            });
   455→   455→            finish();
   456→   456→            return;
   457→   457→          }
   458→   458→          if (!b64) {
   459→   459→            send("error", { message: "Image generation returned nothing — try again." });
   460→   460→            finish();
   461→   461→            return;
   462→   462→          }
   463→   463→
   464→   464→          /* save the picture so it can be displayed + downloaded.
   465→   465→             the API returns jpeg bytes more often than not — sniff the
   466→   466→             real format so content-type never lies (nosniff is set) */
   467→   467→          const isJpeg = /^\/9j\//.test(b64);
   468→   468→          const isPng = /^iVBOR/.test(b64);
   469→   469→          const isWebp = /^UklGR/.test(b64);
   470→   470→          const mime = isPng ? "image/png" : isWebp ? "image/webp" : "image/jpeg";
   471→   471→          const ext = isPng ? "png" : isWebp ? "webp" : "jpg";
   472→   472→          const name = safeName(`memphis-image-${Date.now()}.${ext}`);
   473→   473→          await mkdir(UPLOAD_DIR, { recursive: true });
   474→   474→          const rec = await db.aIFile.create({
   475→   475→            data: {
   476→   476→              sessionId,
   477→   477→              conversationId,
   478→   478→              name,
   479→   479→              mime,
   480→   480→              size: 0,
   481→   481→              kind: "generated",
   482→   482→              path: "",
   483→   483→              status: "processing",
   484→   484→            },
   485→   485→          });
   486→   486→          const filePath = path.join(UPLOAD_DIR, `${rec.id}__${name}`);
   487→   487→          const buf = Buffer.from(b64, "base64");
   488→   488→          await writeFile(filePath, buf);
   489→   489→          const savedFile = await db.aIFile.update({
   490→   490→            where: { id: rec.id },
   491→   491→            data: {
   492→   492→              path: filePath,
   493→   493→              size: buf.length,
   494→   494→              extracted: `AI-${genNote} image from prompt: ${prompt}`,
   495→   495→              status: "ready",
   496→   496→            },
   497→   497→          });
   498→   498→          assistantAtts = attMetas([savedFile]);
   499→   499→
   500→   500→          reply =
   501→   501→            `Here's your image (${genNote}):\n\n` +
   502→   502→            `![${prompt.slice(0, 80)}](/api/ai/files/${rec.id}/raw?sessionId=${encodeURIComponent(sessionId)})\n\n` +
   503→   503→            `Want a different take? Ask me to tweak it — attach this image (or keep it in the chat) and describe the change.`;
   504→   504→          send("delta", { text: reply });
   505→   505→        } else {
   506→   506→          /* ============ WEB SEARCH — plan, search, read, cite ============ */
   507→   507→          let searchContext = "";
   508→   508→          if (webSearch) {
   509→   509→            send("status", { stage: "searching" });
   510→   510→            try {
   511→   511→              const historyText = historyMessages
   512→   512→                .slice(-6)
   513→   513→                .map((m) => `${m.role === "user" ? "User" : "Assistant"}: ${m.content.slice(0, 400)}`)
   514→   514→                .join("\n");
   515→   515→              const planCompletion = (await zai.chat.completions.create({
   516→   516→                messages: [
   517→   517→                  {
   518→   518→                    role: "system",
   519→   519→                    content:
   520→   520→                      "You turn a user's question into web search queries. Reply with ONLY a JSON array of 1-3 short, specific search queries (strings), most relevant first. No prose, no markdown fences.",
   521→   521→                  },
   522→   522→                  {
   523→   523→                    role: "user",
   524→   524→                    content: `Recent conversation:\n${historyText}\n\nQuestion: ${turnText}`,
   525→   525→                  },
   526→   526→                ],
   527→   527→                thinking: { type: "disabled" },

   529→   529→              const planText = planCompletion?.choices?.[0]?.message?.content ?? "";
   530→   530→              const match = planText.match(/\[[\s\S]*\]/);
   531→   531→              const queries = match
   532→   532→                ? (JSON.parse(match[0]) as unknown[]).filter(
   533→   533→                    (q): q is string => typeof q === "string" && q.trim().length > 1
   534→   534→                  ).slice(0, 3)
   535→   535→                : [];
   536→   536→              if (!queries.length) queries.push(turnText.slice(0, 100));
   537→   537→
   538→   538→              const allResults: { url: string; name: string; snippet: string; host: string; date: string }[] = [];
   539→   539→              const seen = new Set<string>();
   540→   540→              await Promise.allSettled(
   541→   541→                queries.map(async (q) => {
   542→   542→                  const res = await zai.functions.invoke("web_search", { query: q, num: 5 });
   543→   543→                  for (const r of res ?? []) {
   544→   544→                    if (!r?.url || seen.has(r.url)) continue;
   545→   545→                    seen.add(r.url);
   546→   546→                    allResults.push({ url: r.url, name: r.name, snippet: r.snippet, host: r.host_name, date: r.date });
   547→   547→                  }
   548→   548→                })

   550→   550→              const top = allResults.slice(0, 6);
   551→   551→
   552→   552→              if (top.length) {

   554→   554→                send("status", { stage: "reading", detail: top[0].host });
   555→   555→                const pages: string[] = [];

   581→   581→          /* ============ PHASE 1 (Reasoning mode) — visible private thinking ============
   582→   582→             The z-ai endpoint has no native reasoning_content channel, so
   583→   583→             Reasoning mode runs a dedicated thinking call first: short
   584→   584→             working notes streamed live into the UI's thinking box, then

   586→   586→          if (mode === "reasoning") {
