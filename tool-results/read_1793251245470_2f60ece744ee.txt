    65→    65→
    66→    66→function Md({ text }: { text: string }) {
    67→    67→  return (
    68→    68→    <div className="memphis-md text-[13.5px] leading-relaxed break-words">
    69→    69→      <ReactMarkdown
    70→    70→        remarkPlugins={[remarkGfm]}
    71→    71→        components={{
    72→    72→          code(props) {
    73→    73→            const { className, children } = props;
    74→    74→            const match = /language-(\w+)/.exec(className || "");
    75→    75→            const code = String(children ?? "").replace(/\n$/, "");
    76→    76→            if (!match && !code.includes("\n")) {
    77→    77→              return (
    78→    78→                <code className="bg-[#161616] border border-[#262626] rounded px-1 py-0.5 text-[12px] font-mono">
    79→    79→                  {code}
    80→    80→                </code>
    81→    81→              );
    82→    82→            }
    83→    83→            return (
    84→    84→              <code className={className}>
    85→    85→                <SyntaxHighlighter
    86→    86→                  style={oneDark as Record<string, Record<string, string>>}
    87→    87→                  language={match ? match[1] : "text"}
    88→    88→                  PreTag="div"
    89→    89→                  customStyle={{
    90→    90→                    borderRadius: "10px",
    91→    91→                    border: "1px solid #1f1f1f",
    92→    92→                    fontSize: "12px",
    93→    93→                    margin: "8px 0",
    94→    94→                    background: "#0a0a0a",
    95→    95→                  }}
    96→    96→                >
    97→    97→                  {code}
    98→    98→                </SyntaxHighlighter>
    99→    99→              </code>
   100→   100→            );
   101→   101→          },
   102→   102→          a(props) {
   103→   103→            const { href, children } = props;
   104→   104→            return (
   105→   105→              <a href={href} target="_blank" rel="noreferrer noopener" className="u-link text-white underline underline-offset-2">
   106→   106→                {children}
   107→   107→              </a>
   108→   108→            );
   109→   109→          },
   110→   110→        }}
   111→   111→      >
   112→   112→        {text}
   113→   113→      </ReactMarkdown>
   114→   114→    </div>
   115→   115→  );
   116→   116→}

   119→   119→
   120→   120→export default function AiView() {
   121→   121→  const [conversations, setConversations] = useState<ConversationRow[]>([]);
   122→   122→  const [activeId, setActiveId] = useState<string | null>(null);
   123→   123→  const [messages, setMessages] = useState<AIMsg[]>([]);
   124→   124→  const [instructions, setInstructions] = useState("");
   125→   125→  const [instructionsDraft, setInstructionsDraft] = useState("");
   126→   126→  const [input, setInput] = useState("");
   127→   127→  const [sending, setSending] = useState(false);
   128→   128→  const [error, setError] = useState("");
   129→   129→  const [cards, setCards] = useState<Flashcard[]>([]);
   130→   130→  const [cardIdx, setCardIdx] = useState(0);
   131→   131→  const [cardFlipped, setCardFlipped] = useState(false);
   132→   132→  const [genCards, setGenCards] = useState(false);
   133→   133→  const [showCards, setShowCards] = useState(false);
   134→   134→  const [savingInstructions, setSavingInstructions] = useState(false);
   135→   135→  const [sidebarOpen, setSidebarOpen] = useState(false);
   136→   136→  const [copiedId, setCopiedId] = useState<string | null>(null);
   137→   137→  const bottomRef = useRef<HTMLDivElement>(null);
   138→   138→  const abortRef = useRef<AbortController | null>(null);
   139→   139→
   140→   140→  /* ---------- conversation list ---------- */
   141→   141→  const loadConversations = useCallback(async () => {
   142→   142→    const sid = aiSessionId();
   143→   143→    if (!sid) return;
   144→   144→    const r = await fetch(`/api/ai/conversations?sessionId=${encodeURIComponent(sid)}`);
   145→   145→    if (r.ok) {
   146→   146→      const data = (await r.json()) as { conversations?: ConversationRow[] };
   147→   147→      setConversations(data.conversations ?? []);
   148→   148→    }
   149→   149→  }, []);
   150→   150→
   151→   151→  useEffect(() => {
   152→   152→    void loadConversations();
   153→   153→    setCards(lsGet<Flashcard[]>(FLASHCARD_KEY, []));
   154→   154→  }, [loadConversations]);
   155→   155→
   156→   156→  /* ---------- load messages for active conversation ---------- */
   157→   157→  useEffect(() => {
   158→   158→    if (!activeId) return;
   159→   159→    const sid = aiSessionId();
   160→   160→    void (async () => {
   161→   161→      const r = await fetch(
   162→   162→        `/api/ai/conversations/${activeId}?sessionId=${encodeURIComponent(sid)}`
   163→   163→      );
   164→   164→      if (r.ok) {
   165→   165→        const data = (await r.json()) as {
   166→   166→          conversation?: ConversationRow;
   167→   167→          messages?: AIMsg[];
   168→   168→        };
   169→   169→        if (data.conversation) {
   170→   170→          setMessages(data.messages ?? []);
   171→   171→          setInstructions(data.conversation.instructions ?? "");
   172→   172→          setInstructionsDraft(data.conversation.instructions ?? "");
   173→   173→        }
   174→   174→      }
   175→   175→    })();
   176→   176→  }, [activeId]);
   177→   177→
   178→   178→  useEffect(() => {
   179→   179→    bottomRef.current?.scrollIntoView({ behavior: "smooth" });
   180→   180→  }, [messages, sending]);
   181→   181→
   182→   182→  /* ---------- actions ---------- */
   183→   183→  const newChat = () => {
   184→   184→    if (sending) abortRef.current?.abort();
   185→   185→    setActiveId(null);
   186→   186→    setMessages([]);
   187→   187→    setInstructions("");
   188→   188→    setInstructionsDraft("");
   189→   189→    setError("");
   190→   190→    setSidebarOpen(false);
   191→   191→  };
   192→   192→
   193→   193→  const send = async () => {
   194→   194→    const text = input.trim();
   195→   195→    if (!text || sending) return;
   196→   196→    const sid = aiSessionId();
   197→   197→    if (!sid) return;
   198→   198→    setError("");
   199→   199→    setInput("");
   200→   200→    setSending(true);
   201→   201→
   202→   202→    const userMsg: AIMsg = { id: "local-" + Date.now(), role: "user", content: text };
   203→   203→    const streamMsg: AIMsg = { id: "stream", role: "assistant", content: "", streaming: true };
   204→   204→    setMessages((m) => [...m, userMsg, streamMsg]);
   205→   205→
   206→   206→    const controller = new AbortController();
   207→   207→    abortRef.current = controller;
   208→   208→
   209→   209→    try {
   210→   210→      const res = await fetch("/api/ai/chat", {
   211→   211→        method: "POST",
   212→   212→        headers: { "content-type": "application/json" },
   213→   213→        body: JSON.stringify({
   214→   214→          sessionId: sid,
   215→   215→          conversationId: activeId,
   216→   216→          message: text,
   217→   217→          instructions: instructionsDraft,
   218→   218→        }),
   219→   219→        signal: controller.signal,
   220→   220→      });
   221→   221→
   222→   222→      if (!res.ok || !res.body) {
   223→   223→        const data = (await res.json().catch(() => ({}))) as { error?: string };
   224→   224→        throw new Error(data.error || `Request failed (${res.status})`);
   225→   225→      }
   226→   226→
   227→   227→      const reader = res.body.getReader();
   228→   228→      const decoder = new TextDecoder();
   229→   229→      let buffer = "";
   230→   230→      let acc = "";
   231→   231→
   232→   232→      for (;;) {
   233→   233→        const { done, value } = await reader.read();
   234→   234→        if (done) break;
   235→   235→        buffer += decoder.decode(value, { stream: true });
   236→   236→        const blocks = buffer.split("\n\n");
   237→   237→        buffer = blocks.pop() ?? "";
   238→   238→        for (const block of blocks) {
   239→   239→          const evLine = block.split("\n").find((l) => l.startsWith("event: "));
   240→   240→          const dataLine = block.split("\n").find((l) => l.startsWith("data: "));
   241→   241→          if (!evLine || !dataLine) continue;
   242→   242→          const event = evLine.slice(7).trim();
   243→   243→          let payload: Record<string, unknown> = {};
   244→   244→          try {
   245→   245→            payload = JSON.parse(dataLine.slice(6));
   246→   246→          } catch {
   247→   247→            continue;
   248→   248→          }
   249→   249→          if (event === "meta" && typeof payload.conversationId === "string") {
   250→   250→            setActiveId(payload.conversationId);
   251→   251→          } else if (event === "delta" && typeof payload.text === "string") {
   252→   252→            acc += payload.text;
   253→   253→            setMessages((m) =>
   254→   254→              m.map((msg) => (msg.id === "stream" ? { ...msg, content: acc } : msg))
   255→   255→            );
   256→   256→          } else if (event === "error") {
   257→   257→            throw new Error(String(payload.message || "The model is unavailable."));
   258→   258→          } else if (event === "done") {
   259→   259→            setMessages((m) =>
   260→   260→              m.map((msg) =>
   261→   261→                msg.id === "stream"
   262→   262→                  ? {
   263→   263→                      id: String(payload.messageId ?? "saved"),
   264→   264→                      role: "assistant",
   265→   265→                      content: acc,
   266→   266→                    }
   267→   267→                  : msg
   268→   268→              )
   269→   269→            );
   270→   270→          }
   271→   271→        }
   272→   272→      }
   273→   273→      void loadConversations();
   274→   274→    } catch (e) {
   275→   275→      const msg = e instanceof Error ? e.message : String(e);
   276→   276→      if (msg !== "The user aborted a request." && !msg.toLowerCase().includes("abort")) {
   277→   277→        setError(msg);
   278→   278→      }
   279→   279→      setMessages((m) => m.filter((x) => x.id !== "stream" || x.content));
   280→   280→      setMessages((m) =>
   281→   281→        m.map((x) => (x.id === "stream" ? { ...x, streaming: false } : x))
   282→   282→      );
   283→   283→    } finally {
   284→   284→      setSending(false);
   285→   285→      abortRef.current = null;
   286→   286→    }
   287→   287→  };
   288→   288→
   289→   289→  const saveInstructions = async () => {
   290→   290→    if (!activeId) {
   291→   291→      setInstructions(instructionsDraft);
   292→   292→      return;
   293→   293→    }
   294→   294→    setSavingInstructions(true);
   295→   295→    await fetch(
   296→   296→      `/api/ai/conversations/${activeId}?sessionId=${encodeURIComponent(aiSessionId())}`,
   297→   297→      {
   298→   298→        method: "PATCH",
   299→   299→        headers: { "content-type": "application/json" },
   300→   300→        body: JSON.stringify({ instructions: instructionsDraft }),
   301→   301→      }
   302→   302→    );
   303→   303→    setInstructions(instructionsDraft);
   304→   304→    setSavingInstructions(false);
   305→   305→  };
   306→   306→
   307→   307→  const deleteConversation = async (id: string) => {
   308→   308→    await fetch(
   309→   309→      `/api/ai/conversations/${id}?sessionId=${encodeURIComponent(aiSessionId())}`,
   310→   310→      { method: "DELETE" }
   311→   311→    );
   312→   312→    if (activeId === id) newChat();
   313→   313→    void loadConversations();
   314→   314→  };
   315→   315→
   316→   316→  const generateFlashcards = async () => {
   317→   317→    if (genCards) return;
   318→   318→    if (!messages.length) {
   319→   319→      setError("Chat about a topic first, then build flashcards from it.");
   320→   320→      return;
   321→   321→    }
   322→   322→    setGenCards(true);
   323→   323→    setError("");
   324→   324→    const r = await fetch("/api/ai/flashcards", {
   325→   325→      method: "POST",
   326→   326→      headers: { "content-type": "application/json" },
   327→   327→      body: JSON.stringify({ sessionId: aiSessionId(), conversationId: activeId }),
   328→   328→    });
   329→   329→    setGenCards(false);
   330→   330→    const data = (await r.json().catch(() => ({}))) as { cards?: Flashcard[]; error?: string };
   331→   331→    if (r.ok && data.cards?.length) {
   332→   332→      const next = [...data.cards, ...cards].slice(0, 120);
   333→   333→      setCards(next);
   334→   334→      lsSet(FLASHCARD_KEY, next);
   335→   335→      setCardIdx(0);
   336→   336→      setCardFlipped(false);
   337→   337→      setShowCards(true);
   338→   338→    } else {
   339→   339→      setError(data.error || "Could not build flashcards — try again.");
   340→   340→    }
   341→   341→  };
   342→   342→
   343→   343→  const copyMessage = async (id: string, content: string) => {
   344→   344→    try {
   345→   345→      await navigator.clipboard.writeText(content);
   346→   346→      setCopiedId(id);
   347→   347→      setTimeout(() => setCopiedId(null), 1500);
   348→   348→    } catch {
   349→   349→      /* clipboard unavailable */
   350→   350→    }
   351→   351→  };
   352→   352→
   353→   353→  /* ---------- render ---------- */
   354→   354→  const card = cards[cardIdx];
   355→   355→
   356→   356→  return (
   357→   357→    <div className="flex-1 flex min-h-0 relative">
   358→   358→      {/* sidebar */}
   359→   359→      <aside
   360→   360→        className={`${sidebarOpen ? "flex" : "hidden"} md:flex absolute md:relative inset-0 md:inset-auto z-30 md:z-auto bg-black/95 md:bg-transparent flex-col w-64 md:w-60 shrink-0 border-r border-[#141414]`}
   361→   361→      >
   362→   362→        <div className="p-3 border-b border-[#141414] flex gap-2">
   363→   363→          <Button
   364→   364→            onClick={newChat}
   365→   365→            className="flex-1 bg-white text-black hover:bg-[#e0e0e0]"
   366→   366→            size="sm"
   367→   367→          >
   368→   368→            <Plus className="size-4 mr-1" /> New chat
   369→   369→          </Button>
   370→   370→          <Button
   371→   371→            variant="outline"
   372→   372→            size="icon"
   373→   373→            className="md:hidden border-[#2a2a2a] text-white"
   374→   374→            onClick={() => setSidebarOpen(false)}
   375→   375→            aria-label="Close conversation list"
   376→   376→          >
   377→   377→            <ChevronRight className="size-4" />
   378→   378→          </Button>
   379→   379→        </div>
   380→   380→        <div className="flex-1 overflow-y-auto p-2 memphis-scroll">
   381→   381→          <p className="mono-label px-2 py-1">Your chats</p>
   382→   382→          {conversations.length === 0 && (
   383→   383→            <p className="text-[12px] text-[#555] px-2 py-2">
   384→   384→              Your conversations appear here — saved on this Memphis server for your browser
   385→   385→              session.
   386→   386→            </p>
   387→   387→          )}
   388→   388→          {conversations.map((c) => (
   389→   389→            <div key={c.id} className="group flex items-center">
   390→   390→              <button
   391→   391→                onClick={() => {
   392→   392→                  setActiveId(c.id);
   393→   393→                  setSidebarOpen(false);
   394→   394→                }}
   395→   395→                className={`flex-1 text-left px-2 py-1.5 rounded-md text-[13px] truncate transition-colors ${
   396→   396→                  activeId === c.id
   397→   397→                    ? "bg-[#141414] text-white"
   398→   398→                    : "text-[#9a9a9a] hover:bg-[#0d0d0d] hover:text-white"
   399→   399→                }`}
   400→   400→              >
   401→   401→                {c.title}
   402→   402→              </button>
   403→   403→              <button
   404→   404→                onClick={() => void deleteConversation(c.id)}
   405→   405→                aria-label={`Delete ${c.title}`}
   406→   406→                className="opacity-0 group-hover:opacity-100 p-1.5 text-[#666] hover:text-white"
   407→   407→              >
   408→   408→                <Trash2 className="size-3.5" />
   409→   409→              </button>
   410→   410→            </div>
   411→   411→          ))}
   412→   412→        </div>
   413→   413→        <div className="p-3 border-t border-[#141414] flex items-center gap-1.5 text-[11px] text-[#777]">
   414→   414→          <InfinityIcon className="size-3.5" />
   415→   415→          GLM · free · unlimited
   416→   416→        </div>
   417→   417→      </aside>
   418→   418→
   419→   419→      {/* main */}
   420→   420→      <div className="flex-1 flex flex-col min-w-0">
   421→   421→        <div className="border-b border-[#141414] px-4 py-2.5 flex items-center gap-2">
   422→   422→          <Button
   423→   423→            variant="ghost"
   424→   424→            size="icon"
   425→   425→            className="md:hidden text-white size-8"
   426→   426→            onClick={() => setSidebarOpen(true)}
   427→   427→            aria-label="Open conversation list"
   428→   428→          >
   429→   429→            <Menu className="size-4" />
   430→   430→          </Button>
   431→   431→          <Sparkles className="size-4 text-white" />

   434→   434→            powered by GLM
   435→   435→          </span>
   436→   436→          <div className="ml-auto flex items-center gap-2">

   439→   439→              size="sm"
   440→   440→              onClick={() => void generateFlashcards()}
   441→   441→              disabled={genCards}
   442→   442→              className="border-[#2a2a2a] text-white hover:bg-[#101010]"
   443→   443→            >

   446→   446→            </Button>
   447→   447→            {cards.length > 0 && (
   448→   448→              <Button
   449→   449→                variant="ghost"
   450→   450→                size="sm"
   451→   451→                onClick={() => {
   452→   452→                  setCardIdx(0);
   453→   453→                  setCardFlipped(false);
   454→   454→                  setShowCards(true);
   455→   455→                }}
   456→   456→                className="text-[#9a9a9a] hover:text-white"
   457→   457→              >
   458→   458→                Study ({cards.length})
   459→   459→              </Button>
   460→   460→            )}
   461→   461→          </div>
   462→   462→        </div>
   463→   463→
   464→   464→        {/* messages */}
   465→   465→        <div className="flex-1 overflow-y-auto memphis-scroll">
   466→   466→          <div className="max-w-2xl mx-auto px-4 py-6 space-y-4">
   467→   467→            {messages.length === 0 && !sending && (
   468→   468→              <div className="text-center py-12">
   469→   469→                <Sparkles className="size-7 mx-auto text-[#555] mb-3" />
   470→   470→                <h2 className="text-lg text-white font-semibold">How can I help today?</h2>
   471→   471→                <p className="text-[13px] text-[#777] mt-2">
   472→   472→                  A real AI chat running on GLM — free, unlimited, no account needed.
   473→   473→                  Ask anything.
   474→   474→                </p>
   475→   475→                <div className="mt-6 grid sm:grid-cols-2 gap-2 text-left">
   476→   476→                  {[
   477→   477→                    "Explain how a scramjet engine works",
   478→   478→                    "Write a Python script to rename my files",
   479→   479→                    "Help me draft a message to my teacher",
   480→   480→                    "Give me a 20-minute full-body workout plan",
