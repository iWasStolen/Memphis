   131→   131→/* split a message into TTS-sized chunks (API limit is 1024 chars) */
   132→   132→function splitForTts(text: string, max = 950): string[] {
   133→   133→  const clean = text
   134→   134→    .replace(/```[\s\S]*?```/g, " — code block omitted — ")
   135→   135→    .replace(/!\[[^\]]*\]\([^)]*\)/g, " — image — ");
   136→   136→  const sentences = clean.match(/[^.!?\n]+[.!?]*\s*/g) ?? [clean];
   137→   137→  const chunks: string[] = [];
   138→   138→  let cur = "";
   139→   139→  for (const s of sentences) {
   140→   140→    if ((cur + s).length > max) {
   141→   141→      if (cur.trim()) chunks.push(cur);
   142→   142→      if (s.length > max) {
   143→   143→        for (let i = 0; i < s.length; i += max) chunks.push(s.slice(i, i + max));
   144→   144→        cur = "";
   145→   145→      } else cur = s;
   146→   146→    } else cur += s;
   147→   147→  }
   148→   148→  if (cur.trim()) chunks.push(cur);
   149→   149→  return chunks;
   150→   150→}

   152→   152→/* 16 kHz mono PCM WAV encoder (for voice input) */
   153→   153→function encodeWav(buffer: AudioBuffer, targetRate = 16000): ArrayBuffer {
   154→   154→  const src = buffer.getChannelData(0);
   155→   155→  const ratio = buffer.sampleRate / targetRate;
   156→   156→  const length = Math.max(1, Math.floor(src.length / ratio));
   157→   157→  const data = new Float32Array(length);
   158→   158→  for (let i = 0; i < length; i++) {
   159→   159→    const pos = i * ratio;
   160→   160→    const i0 = Math.floor(pos);
   161→   161→    const i1 = Math.min(i0 + 1, src.length - 1);
   162→   162→    const frac = pos - i0;
   163→   163→    data[i] = src[i0] * (1 - frac) + src[i1] * frac;
   164→   164→  }
   165→   165→  const out = new ArrayBuffer(44 + length * 2);
   166→   166→  const view = new DataView(out);
   167→   167→  const wstr = (off: number, s: string) => {
   168→   168→    for (let i = 0; i < s.length; i++) view.setUint8(off + i, s.charCodeAt(i));
   169→   169→  };
   170→   170→  wstr(0, "RIFF");
   171→   171→  view.setUint32(4, 36 + length * 2, true);
   172→   172→  wstr(8, "WAVE");
   173→   173→  wstr(12, "fmt ");
   174→   174→  view.setUint32(16, 16, true);
   175→   175→  view.setUint16(20, 1, true);
   176→   176→  view.setUint16(22, 1, true);
   177→   177→  view.setUint32(24, targetRate, true);
   178→   178→  view.setUint32(28, targetRate * 2, true);
   179→   179→  view.setUint16(32, 2, true);
   180→   180→  view.setUint16(34, 16, true);
   181→   181→  wstr(36, "data");
   182→   182→  view.setUint32(40, length * 2, true);
   183→   183→  let off = 44;
   184→   184→  for (let i = 0; i < length; i++, off += 2) {
   185→   185→    const s = Math.max(-1, Math.min(1, data[i]));
   186→   186→    view.setInt16(off, s < 0 ? s * 0x8000 : s * 0x7fff, true);
   187→   187→  }
   188→   188→  return out;
   189→   189→}

   194→   194→  return (
   195→   195→    <div className="memphis-md text-[13.5px] leading-relaxed break-words">

   201→   201→            const match = /language-(\w+)/.exec(className || "");
   202→   202→            const code = String(children ?? "").replace(/\n$/, "");
   203→   203→            if (!match && !code.includes("\n")) {

   206→   206→                  {code}

   211→   211→              <code className={className}>
   212→   212→                <SyntaxHighlighter
   213→   213→                  style={oneDark as Record<string, Record<string, string>>}
   214→   214→                  language={match ? match[1] : "text"}
   215→   215→                  PreTag="div"
   216→   216→                  customStyle={{
   217→   217→                    borderRadius: "10px",
   218→   218→                    border: "1px solid #1f1f1f",

   222→   222→                  }}

   224→   224→                  {code}
   225→   225→                </SyntaxHighlighter>
   226→   226→              </code>
   227→   227→            );
   228→   228→          },
   229→   229→          a(props) {
   230→   230→            const { href, children } = props;
   231→   231→            return (
   232→   232→              <a href={href} target="_blank" rel="noreferrer noopener" className="u-link text-white underline underline-offset-2">
   233→   233→                {children}
   234→   234→              </a>
   235→   235→            );
   236→   236→          },
   237→   237→          img(props) {

   241→   241→              <img
   242→   242→                src={src}
   243→   243→                alt={alt ?? ""}

   247→   247→              />
   248→   248→            );
   249→   249→          },
   250→   250→        }}
   251→   251→      >
   252→   252→        {text}
   253→   253→      </ReactMarkdown>
   254→   254→    </div>
   255→   255→  );
   256→   256→}
